FILE:src/sldb/code/lang_sql.py
"""SQL (`.sql`, PostgreSQL DDL and migrations) into facts of the same form as Python's and TS's:
the module, then each table, view, index and key in source order, a table's columns and keys right
after it (readers in `sql_readers`, `sql_columns` and `sql_keys`; facts in `sql_fact`).

The module's qualified name is its path without extension, like `py_module_name`. An object's
qualified name is its name as written without double quotes (`public.users` or `users`), a column's
is `<table>.<column>`, a key's `<table>.<constraint>`; statements inside `BEGIN; ... COMMIT;` count,
and a statement with a local parse error keeps its object and the columns whose name reads. A
column the grammar does not parse whole (Django's `id ... GENERATED BY DEFAULT AS IDENTITY`,
tree-sitter-sql 0.3.13) is kept with `attrs.parse_partial`. `ALTER TABLE` is read only for its
`ADD CONSTRAINT name {PRIMARY KEY|FOREIGN KEY|UNIQUE}`. A foreign key written without referenced
columns (`REFERENCES public.parent`) has `attrs.references.columns: []`: the referenced table's
primary key. Out: any other `ALTER TABLE` action, a
column-level `REFERENCES` (no fact, no mark), `CHECK` and `EXCLUDE`, other dialects, and any state
across migrations: each file is read on its own. An object repeated in the same file takes `~2`,
`~3` from `Ids`, and its columns hang from the suffixed name. The anchor symbol is that qualified
name. `imports` is always empty. Since local errors are tolerated, the module's `has_error` (which
makes an anchor `unparseable`) is set only when the file has errors and no statement reads;
`attrs.parse_errors` keeps the raw `has_error`.
"""

from __future__ import annotations

import hashlib
from pathlib import Path
from typing import Any

from sldb.code.facts import fact, rels
from sldb.code.ids import Ids
from sldb.code.imports import py_module_name
from sldb.code.runtime import parse
from sldb.code.sql_columns import column_attrs
from sldb.code.sql_fact import key_facts, link_references, sql_fact
from sldb.code.sql_fingerprints import sql_fingerprints
from sldb.code.sql_keys import alter_keys, table_keys
from sldb.code.sql_readers import KINDS, columns, creations, sql_doc, sql_name


def scan_sql_file(rel: str, root: Path, idgen: Ids) -> list[dict[str, Any]]:
    src = (root / rel).read_bytes()
    rootn, module = parse(src, rel).root_node, py_module_name(rel)
    ctx = {"rel": rel, "module": module, "src": src, "digest": "sha256:" + hashlib.sha256(src).hexdigest(), "mid": idgen.take(module or "__init__"),
           "ids": idgen, "local": Ids(), "tables": {}}
    found = creations(rootn)
    facts = [_module_fact(rootn, ctx, rootn.has_error and not found)]
    for stmt, node, name in found:
        facts += (_alter_facts if node.type == "alter_table" else _object_facts)(stmt, node, name, ctx)
    return link_references(facts)


def _module_fact(rootn: Any, ctx: dict[str, Any], unreadable: bool) -> dict[str, Any]:
    """`has_error` (an anchor's `unparseable`) only when nothing reads; `parse_errors` says whether the tree has any error."""
    module, nlines = ctx["module"], len(ctx["src"].decode("utf-8", "replace").splitlines())
    return fact(id=ctx["mid"], kind="module", name=module.rsplit(".", 1)[-1] if module else "__init__", qualified_name=module, path=ctx["rel"],
                span={"start_line": 1, "end_line": max(nlines, 1)}, source_hash=ctx["digest"], doc=None, relations=rels(None, []),
                attrs={"native_kind": rootn.type, "module": module, "imports": [], "nested_imports": [], "has_error": unreadable, "parse_errors": rootn.has_error},
                fingerprints=sql_fingerprints(rootn, None, None, ctx["src"]))


def _object_facts(stmt: Any, node: Any, name: Any, ctx: dict[str, Any]) -> list[dict[str, Any]]:
    """The table, view or index, then (for a table) its columns and its keys, contained in it."""
    src = ctx["src"]
    obj = sql_fact(node, name, KINDS[node.type], ctx["local"].take(sql_name(name, src)), sql_doc(stmt, src), ctx["mid"], ctx)
    if obj["kind"] == "table":
        ctx["tables"].setdefault(sql_name(name, src), obj["id"])
    cols = [_column_fact(col, cname, obj, ctx) for col, cname in columns(node)]
    return [obj] + cols + key_facts(table_keys(node), obj["qualified_name"], obj["id"], ctx, named=False)


def _column_fact(col: Any, cname: Any, obj: dict[str, Any], ctx: dict[str, Any]) -> dict[str, Any]:
    out = sql_fact(col, cname, "column", f"{obj['qualified_name']}.{sql_name(cname, ctx['src'])}", None, obj["id"], ctx)
    out["attrs"].update(column_attrs(col, cname, ctx["src"]))
    return out


def _alter_facts(stmt: Any, node: Any, name: Any, ctx: dict[str, Any]) -> list[dict[str, Any]]:
    """The keys of an `ALTER TABLE ... ADD CONSTRAINT name`, under the table read before it, or the module."""
    table = sql_name(name, ctx["src"])
    return key_facts(alter_keys(node), table, ctx["tables"].get(table, ctx["mid"]), ctx, named=True)

FILE:src/sldb/code/sql_readers.py
"""Readers over tree-sitter-sql nodes (PostgreSQL DDL): the `CREATE` statements, their names, columns and `--` comments.

`CREATE TABLE`, `CREATE VIEW`, `CREATE MATERIALIZED VIEW`, `CREATE INDEX` and `ALTER TABLE` with an
`ADD CONSTRAINT` (its keys are read in `sql_keys`; any other `ALTER TABLE` is left out) are read, at the top
level and inside `BEGIN; ... COMMIT;` (a `transaction` node), as `sqlmigrate` of Django writes them.
The name keeps its schema when written (`public.users`) and loses its double quotes: `"app_book"`
and `app_book` are the same name, also for an index (whose name the grammar gives as a `literal`).
Reading is tolerant: a statement with a local parse error still gives its object when the name
reads cleanly, and each column whose `column_definition` starts with a clean name, even with an error
inside: tree-sitter-sql 0.3.13 does not parse `GENERATED BY DEFAULT AS IDENTITY` (Django's primary key
on Postgres), and that `id` is read from its text (`sql_columns`). A statement whose name does not
read gives no fact at all. Table constraints (`PRIMARY KEY (...)`, `FOREIGN KEY`, `UNIQUE` at the end) are not columns.
"""

from __future__ import annotations

from typing import Any

from sldb.code.readers import text
from sldb.code.sql_columns import column_name

KINDS = {"create_table": "table", "create_view": "view", "create_materialized_view": "view", "create_index": "index"}
_NAMES = {"create_index": ("identifier", "literal")}


def creations(root: Any) -> list[tuple[Any, Any, Any]]:
    """`(statement, node, name node)` of each `CREATE` and each `ALTER TABLE ... ADD CONSTRAINT` whose name reads, in source order, also inside transactions."""
    out = []
    for stmt in root.named_children:
        if stmt.type == "transaction":
            out += creations(stmt)
        for node in (n for n in stmt.named_children if stmt.type == "statement" and _read(n)):
            name = next((c for c in node.named_children if c.type in _NAMES.get(node.type, ("object_reference",))), None)
            out += [(stmt, node, name)] if name is not None and not name.has_error and name.end_byte > name.start_byte else []
    return out


def _read(node: Any) -> bool:
    return node.type in KINDS or (node.type == "alter_table" and any(c.type == "add_constraint" for c in node.named_children))


def columns(node: Any) -> list[tuple[Any, Any]]:
    """`(column_definition, name node)` of a `CREATE TABLE`, in order, also a definition with a parse error whose name reads first."""
    defs = next((c for c in node.named_children if c.type == "column_definitions"), None)
    kids = [c for c in (defs.named_children if defs is not None else []) if c.type == "column_definition"]
    return [(c, column_name(c)) for c in kids if column_name(c) is not None]


def sql_name(node: Any, src: bytes) -> str:
    """The name as written, without double quotes: `"public"."users"` is `public.users`."""
    return text(node, src).replace('"', "")


def sql_doc(stmt: Any, src: bytes) -> str | None:
    """The `--` lines right above `stmt`, contiguous and each on its own line, joined by newlines without `--`."""
    lines, cur, row = [], stmt.prev_sibling, stmt.start_point[0]
    while cur is not None and cur.type == "comment" and text(cur, src).startswith("--") and cur.end_point[0] + 1 == row and _own_line(cur):
        lines.insert(0, text(cur, src)[2:].strip())
        row, cur = cur.start_point[0], cur.prev_sibling
    return "\n".join(lines).strip() or None


def _own_line(comment: Any) -> bool:
    prev = comment.prev_sibling
    return prev is None or prev.end_point[0] < comment.start_point[0]


def sql_norm(node: Any, src: bytes) -> str:
    """The node's text with whitespace collapsed (no requoting: in SQL `"x"` is an identifier)."""
    return " ".join(text(node, src).split())
